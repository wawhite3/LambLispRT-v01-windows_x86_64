733class LL_String {
734public:
735
736 LL_String() { _init(); set(""); }
737 LL_String(const char *other) { _init(); set(other); }
738 LL_String(const char *other, int n) { _init(); set(other, n); }
739 LL_String(const LL_String &other) { _init(); set(other.ptr, other.len); }
740
741 ~LL_String() { if (ptr) { delete[] ptr; ptr = 0; } len = cap = 0; }
742
743 void set(const char *s) { set(s, s ? (int) strlen(s) : 0); }
744
745 void set(const char *s, int n) {
746 if (!s || n <= 0) { _ensure(0); ptr[0] = '\0'; len = 0; return; }
747 if (ptr && s >= ptr && s <= ptr + len) {
748 const int off = (int) (s - ptr);
749 memmove(ptr, ptr + off, (size_t) n);
750 }
751 else {
752 _ensure(n);
753 memmove(ptr, s, (size_t) n);
754 }
755 ptr[n] = '\0';
756 len = n;
757 }
758
759 LL_String concat(const char *s1, const char *s2) {
760 LL_String r;
761 const int n1 = s1 ? (int) strlen(s1) : 0;
762 const int n2 = s2 ? (int) strlen(s2) : 0;
763 r._ensure(n1 + n2);
764 if (n1) memmove(r.ptr, s1, (size_t) n1);
765 if (n2) memmove(r.ptr + n1, s2, (size_t) n2);
766 r.ptr[n1 + n2] = '\0';
767 r.len = n1 + n2;
768 return r;
769 }
770
771 LL_String &append(const char *s1) {
772 if (s1 && *s1) {
773 const int n1 = (int) strlen(s1);
774 const bool self = (ptr && s1 >= ptr && s1 <= ptr + len);
775 const int off = self ? (int) (s1 - ptr) : 0;
776 _ensure(len + n1);
777 memmove(ptr + len, self ? ptr + off : s1, (size_t) n1);
778 len += n1;
779 ptr[len] = '\0';
780 }
781 return *this;
782 }
783
786 LL_String &append(char c) {
787 _ensure(len + 1);
788 ptr[len++] = c;
789 ptr[len] = '\0';
790 return *this;
791 }
792
793 LL_String substring(int start, int len_) { return LL_String(&(ptr[start]), len_); }
794
795 LL_String &operator=(const char *other) { set(other); return *this; }
796 LL_String &operator=(const LL_String &other) { if (this != &other) set(other.ptr, other.len); return *this; }
797
798 bool operator==(const char *other) { return strcmp(ptr, other) == 0; }
799 bool operator!=(const char *other) { return strcmp(ptr, other) != 0; }
800 bool operator==(const LL_String &other) { return strcmp(ptr, other.ptr) == 0; }
801 bool operator!=(const LL_String &other) { return strcmp(ptr, other.ptr) != 0; }
802 char &operator[](int ix) { return ptr[ix]; }
803
804 LL_String operator+(const char *other) { return concat(ptr, other); }
805 LL_String operator+(const LL_String &other) { return concat(ptr, other.ptr); }
806 LL_String operator+(char c) { char cstr[2]; cstr[0] = c; cstr[1] = '\0'; return concat(ptr, cstr); }
807
808 LL_String &operator+=(const char *other) { return append(other); }
809 LL_String &operator+=(const LL_String &other) { return append(other.ptr); }
810 LL_String &operator+=(char c) { return append(c); }
811
812 char *c_str() const { return ptr; }
813 int length() const { return len; }
814
815private:
816 void _init() { ptr = 0; len = 0; cap = 0; }
817
820 void _ensure(int need) {
821 if (cap >= need + 1) return;
822 int ncap = cap ? cap : 16;
823 while (ncap < need + 1) ncap += ncap;
825 char *np = new char[ncap];
826 if (ptr) { memmove(np, ptr, (size_t) len + 1); delete[] ptr; }
827 else { np[0] = '\0'; }
828 ptr = np;
829 cap = ncap;
830 }
831
832 char *ptr;
833 int len;
834 int cap;
835};

1216inline int ll_u8_seqlen(unsigned char b0)
1217{
1218 if (b0 < 0x80) return 1;
1219 if ((b0 & 0xE0) == 0xC0) return 2;
1220 if ((b0 & 0xF0) == 0xE0) return 3;
1221 if ((b0 & 0xF8) == 0xF0) return 4;
1222 return 1;
1223}
1224
1227inline ll_codepoint_t ll_u8_decode(const char *p, int *nb)
1228{
1229 unsigned char b0 = (unsigned char) p[0];
1230 int n = ll_u8_seqlen(b0);
1231 if (nb) *nb = n;
1232 if (n == 1) return (ll_codepoint_t) b0;
1233 ll_codepoint_t cp = b0 & (0xFF >> (n + 1));
1234 for (int i = 1; i < n; i++) {
1235 unsigned char c = (unsigned char) p[i];
1236 if ((c & 0xC0) != 0x80) { if (nb) *nb = 1; return (ll_codepoint_t) b0; }
1237 cp = (cp << 6) | (c & 0x3F);
1238 }
1239 return cp;
1240}
1241
1243inline int ll_u8_encode(ll_codepoint_t cp, char *out)
1244{
1245 if (cp < 0x80) { out[0] = (char) cp; return 1; }
1246 if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
1247 if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
1248 out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
1249 out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
1250 out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
1251}

1313 char *dec(Word_t n) {
1314 char *ptr = slot_end();
1315
1316 *ptr = '\0';
1317 do {
1318 Word_t d = n % 10;
